// Workers AI Image Generation — available as IMAGE_AI in every workspace.
// No OAuth or external credentials required.

export interface GenerateImageOptions {
  model?: string;  // default: "@cf/black-forest-labs/flux-1-schnell"
  steps?: number;
}

export interface GeneratedImage {
  data: string;        // base64-encoded PNG (no data: prefix)
  mimeType: "image/png";
  prompt: string;
}

/**
 * Session for generating images with Workers AI.
 *
 * PREFERRED USAGE — display inline in chat with no Gadget:
 *
 *   export default async function(self, env, ctx) {
 *     const url = await env.IMAGE_AI.generateImageUrl("a sailboat at sunset");
 *     console.log(`![generated image](${url})`);
 *   }
 *
 * generateImageUrl() stores the image in Cloudflare and returns a short URL.
 * The markdown image tag renders inline in the chat message.
 * URL expires after 1 hour.
 *
 * DO NOT use console.log with base64 data — it exceeds the log size limit.
 * DO NOT create a Gadget just to display a single image — use generateImageUrl() instead.
 * Only create a Gadget if the user explicitly wants an interactive image tool.
 */
export interface ImageSession {
  /** Returns a short URL to the generated image. Log it as a markdown image tag to display inline. */
  generateImageUrl(prompt: string, options?: GenerateImageOptions): Promise<string>;

  /** Returns raw base64 PNG data. Use only when you need the bytes directly (e.g. for another API). */
  generateImage(prompt: string, options?: GenerateImageOptions): Promise<GeneratedImage>;
}
