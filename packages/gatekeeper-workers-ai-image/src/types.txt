// Workers AI Image Generation — agent-facing TypeScript declarations.
// IMAGE_AI is automatically available as an ambient binding in every workspace.
// No OAuth or external credentials required.

/** Options for generating an image. */
export interface GenerateImageOptions {
  /**
   * Workers AI text-to-image model. Defaults to "@cf/black-forest-labs/flux-1-schnell".
   * Other supported model: "@cf/stabilityai/stable-diffusion-xl-base-1.0"
   */
  model?: string;
  /**
   * Number of inference steps. Higher values improve quality at the cost of latency.
   * Omit to use the model's default (4 for flux-1-schnell).
   */
  steps?: number;
}

/** A generated image. */
export interface GeneratedImage {
  /** Base64-encoded PNG image data (no data: prefix). */
  data: string;
  /** Always "image/png". */
  mimeType: "image/png";
  /** The exact prompt used to generate this image. */
  prompt: string;
}

/**
 * Session for generating images with Workers AI.
 *
 * IMPORTANT — displaying images:
 * executeCode() does not expose return values to the chat — only console.log output
 * is visible. To display a generated image inline in the conversation WITHOUT creating
 * a Gadget, console.log a markdown image tag:
 *
 *   export default async function(self, env, ctx) {
 *     const img = await env.IMAGE_AI.generateImage("a sailboat at sunset");
 *     console.log(`![${img.prompt}](data:image/png;base64,${img.data})`);
 *   }
 *
 * The markdown image tag will render inline in the chat message. No Gadget needed.
 *
 * Only create a Gadget if the user explicitly wants an interactive tool (e.g. to
 * re-generate with different prompts, or to display multiple images side-by-side).
 */
export interface ImageSession {
  /**
   * Generate an image from a text prompt using Workers AI.
   * Returns base64-encoded PNG data in result.data.
   *
   * To display inline in chat: console.log(`![desc](data:image/png;base64,${img.data})`)
   */
  generateImage(prompt: string, options?: GenerateImageOptions): Promise<GeneratedImage>;
}
